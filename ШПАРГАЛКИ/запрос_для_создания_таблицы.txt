create table имя_таблицы (
	имя_первичного_ключа serial primary key,
	имя_столбца тип_данных [параметры],
	имя_столбца тип_данных [параметры],
	имя_внешнего_ключа тип_данных references другая_таблица(первичный_ключ) on delete cascade on update cascade
);


ТИПЫ ДАННЫХ:
int – целые числа
serial – автоматически увеличивающиеся целые числа
double – вещественные
decimal – число с переменной длиной и высокой точностью
money – для денег
varchar(n) – строка с максимальным количеством символов n
text – строка переменной длины без максимального количества символов
bool - логический тип
date, timestamp, time – дата и время


ПАРАМЕТРЫ:
unique – уникальное значение
not null – непустое
default ‘default value’ – значение по умолчанию


** [] - необязательные параметры
